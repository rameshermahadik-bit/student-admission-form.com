<script>
    document.getElementById('admissionForm').addEventListener('submit', async function(e) {
        e.preventDefault();
        
        const statusDiv = document.getElementById('statusMessage');
        statusDiv.style.color = 'blue';
        statusDiv.innerText = 'Submitting data to Power Automate...';

        // Collect Form Data
        const formData = {
            studentId: document.getElementById('studentId').value,
            studentName: document.getElementById('studentName').value,
            dob: document.getElementById('dob').value,
            gender: document.querySelector('input[name="gender"]:checked').value,
            mobile: document.getElementById('mobile').value,
            email: document.getElementById('email').value,
            course: document.getElementById('course').value
        };

        // REPLACE THIS WITH YOUR POWER AUTOMATE HTTP POST URL
        const powerAutomateUrl = "YOUR_POWER_AUTOMATE_WEBHOOK_URL_HERE";

        try {
            const response = await fetch(powerAutomateUrl, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(formData)
            });

            if (response.ok) {
                statusDiv.style.color = 'green';
                statusDiv.innerText = 'Form submitted successfully & Saved to Excel!';
                document.getElementById('admissionForm').reset();
            } else {
                statusDiv.style.color = 'red';
                statusDiv.innerText = 'Error submitting form. Please check Power Automate Flow.';
            }
        } catch (error) {
            console.error('Error:', error);
            statusDiv.style.color = 'red';
            statusDiv.innerText = 'Network error or URL missing!';
        }
    });
</script>
